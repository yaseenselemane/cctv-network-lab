! R1 - routes between the three VLANs and enforces who may reach the CCTV VLAN
hostname R1
no ip domain-lookup
interface g0/0
 description Trunk to SW1
 no shutdown
interface g0/0.10
 description STAFF
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
 ip access-group STAFF-IN in
interface g0/0.20
 description CCTV
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
 ip access-group CCTV-IN in
interface g0/0.30
 description SECURITY
 encapsulation dot1Q 30
 ip address 192.168.30.1 255.255.255.0
 ip access-group SECURITY-IN in
interface g0/1
 shutdown
!
! DHCP for people; cameras and the NVR are static
ip dhcp excluded-address 192.168.10.1 192.168.10.9
ip dhcp excluded-address 192.168.30.1 192.168.30.9
ip dhcp pool STAFF
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
ip dhcp pool SECURITY
 network 192.168.30.0 255.255.255.0
 default-router 192.168.30.1
!
ip access-list extended STAFF-IN
 remark Staff may not reach the CCTV VLAN at all
 deny ip 192.168.10.0 0.0.0.255 192.168.20.0 0.0.0.255
 permit ip any any
ip access-list extended SECURITY-IN
 remark Security office may view the NVR web page, nothing else in CCTV
 permit tcp 192.168.30.0 0.0.0.255 host 192.168.20.10 eq 80
 permit icmp 192.168.30.0 0.0.0.255 host 192.168.20.10 echo
 deny ip 192.168.30.0 0.0.0.255 192.168.20.0 0.0.0.255
 permit ip any any
ip access-list extended CCTV-IN
 remark Cameras and NVR may only answer the security office
 permit tcp host 192.168.20.10 eq 80 192.168.30.0 0.0.0.255 established
 permit icmp host 192.168.20.10 192.168.30.0 0.0.0.255 echo-reply
 permit icmp 192.168.20.0 0.0.0.255 host 192.168.20.1
 deny ip any any
